      <form id="bookForm" class="space-y-3">
        <input type="text" id="title" placeholder="Titolo" required class="w-full p-2.5 rounded-xl border border-stone-200 text-sm focus:outline-none focus:border-stone-500">
        <input type="text" id="author" placeholder="Autore" required class="w-full p-2.5 rounded-xl border border-stone-200 text-sm focus:outline-none focus:border-stone-500">
        <div class="grid grid-cols-2 gap-2">
          <input type="number" id="pages" placeholder="Pagine totali" required class="p-2.5 rounded-xl border border-stone-200 text-sm focus:outline-none focus:border-stone-500">
          <input type="text" id="genre" placeholder="Genere (es. Romanzo)" required class="p-2.5 rounded-xl border border-stone-200 text-sm focus:outline-none focus:border-stone-500">
        </div>

        <button type="submit" class="w-full bg-stone-900 text-white font-semibold py-2.5 rounded-xl text-sm active:scale-[0.99] transition">Aggiungi Libro</button>

  <script>
    const GOAL_ANNUALE = 12;
    const currentYear = new Date().getFullYear();
    document.getElementById('currentYear').textContent = currentYear;

    let books = JSON.parse(localStorage.getItem('myReadingList') || '[]');

    function save() {
      localStorage.setItem('myReadingList', JSON.stringify(books));
      render();
    }

    function addBook(e) {
      e.preventDefault();
      const newBook = {
        id: Date.now(),
        title: document.getElementById('title').value,
        author: document.getElementById('author').value,
        pages: parseInt(document.getElementById('pages').value),
        currentPage: 0,
        genre: document.getElementById('genre').value.trim(),
        cover: document.getElementById('cover').value || 'https://images.unsplash.com/photo-1544947950-fa07a98d237f?w=300&q=80',
        completedYear: null,
        rating: 0
      };
      books.push(newBook);
      save();
      e.target.reset();
    }

    function updatePage(id, delta) {
      const b = books.find(item => item.id === id);
      if (!b) return;
      b.currentPage = Math.max(0, Math.min(b.pages, b.currentPage + delta));
      if (b.currentPage === b.pages && !b.completedYear) {
        b.completedYear = currentYear;
      } else if (b.currentPage < b.pages) {
        b.completedYear = null;
      }
      save();
    }

    function setRating(id, stars) {
      const b = books.find(item => item.id === id);
      if (b) {
        b.rating = stars;
        save();
      }
    }

    function deleteBook(id) {
      if (confirm("Vuoi rimuovere questo libro?")) {
        books = books.filter(b => b.id !== id);
        save();
      }
    }

    function render() {
      const readingEl = document.getElementById('readingList');
      const completedEl = document.getElementById('completedList');
      readingEl.innerHTML = '';
      completedEl.innerHTML = '';

      const completedThisYear = books.filter(b => b.completedYear === currentYear);
      
      // Statistiche
      document.getElementById('goalText').textContent = `${completedThisYear.length} / ${GOAL_ANNUALE} completati`;
      document.getElementById('goalBar').style.width = `${Math.min(100, (completedThisYear.length / GOAL_ANNUALE) * 100)}%`;

      const totalPages = completedThisYear.reduce((acc, b) => acc + b.pages, 0) + 
                          books.filter(b => !b.completedYear).reduce((acc, b) => acc + b.currentPage, 0);
      document.getElementById('totalPagesRead').textContent = totalPages.toLocaleString();

      // Calcolo Genere Preferito (basato sui voti o sui più letti tra i completati)
      const genreCounts = {};
      completedThisYear.forEach(b => {
        genreCounts[b.genre] = (genreCounts[b.genre] || 0) + 1;
      });
      const topGenre = Object.keys(genreCounts).sort((a, b) => genreCounts[b] - genreCounts[a])[0] || '-';
      document.getElementById('topGenre').textContent = topGenre;

      // Render Cards
      books.forEach(book => {
        const percent = Math.round((book.currentPage / book.pages) * 100);
        const card = document.createElement('div');
        card.className = "bg-white p-3.5 rounded-2xl shadow-sm border border-stone-200/80 flex gap-3.5 items-center";

        card.innerHTML = `
          <img src="${book.cover}" class="w-16 h-22 object-cover rounded-lg bg-stone-200 shrink-0">
          <div class="flex-1 min-w-0">
            <h4 class="font-bold text-stone-900 text-sm truncate">${book.title}</h4>
            <p class="text-xs text-stone-500 mb-2 truncate">${book.author} • <span class="bg-stone-100 px-1.5 py-0.5 rounded">${book.genre}</span></p>
            
            ${!book.completedYear ? `
              <div class="flex justify-between items-center text-xs mb-1">
                <span class="font-semibold text-amber-700">${percent}%</span>
                <span class="text-stone-400">${book.currentPage}/${book.pages} pag.</span>
              </div>
              <div class="w-full bg-stone-100 h-1.5 rounded-full overflow-hidden mb-2">
                <div class="bg-amber-600 h-full rounded-full" style="width: ${percent}%"></div>
              </div>
              <div class="flex gap-1">
                <button onclick="updatePage(${book.id}, -10)" class="text-xs bg-stone-100 px-2 py-1 rounded-md font-semibold">-10</button>
                <button onclick="updatePage(${book.id}, 10)" class="text-xs bg-stone-100 px-2 py-1 rounded-md font-semibold">+10</button>
                <button onclick="updatePage(${book.id}, ${book.pages - book.currentPage})" class="text-xs bg-amber-100 text-amber-800 px-2 py-1 rounded-md font-semibold ml-auto">Finito</button>
              </div>
            ` : `
              <div class="flex items-center gap-1 text-xs">
                <span class="text-stone-400">Voto:</span>
                ${[1,2,3,4,5].map(star => `
                  <span onclick="setRating(${book.id}, ${star})" class="cursor-pointer ${book.rating >= star ? 'text-amber-500' : 'text-stone-200'}">★</span>
                `).join('')}
              </div>
              <button onclick="deleteBook(${book.id})" class="text-[11px] text-red-500 mt-2 hover:underline">Rimuovi</button>
            `}
          </div>
        `;

        if (book.completedYear) {
          completedEl.appendChild(card);
        } else {
          readingEl.appendChild(card);
        }
      });
    }

    function exportData() {
      const blob = new Blob([JSON.stringify(books, null, 2)], { type: 'application/json' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = `letture_backup_${currentYear}.json`;
      a.click();
    }

    function importData(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (ev) => {
        books = JSON.parse(ev.target.result);
        save();
      };
      reader.readAsText(file);
    }

    document.getElementById('bookForm').addEventListener('submit', addBook);
    render();
  </script>
